export function findVoicings(symbol,settings={}){
 const chord=parseChord(symbol);if(!chord)return {chord:null,results:[],omitted:[],reason:'Símbolo não reconhecido. Confira a grafia ou use um dos exemplos.'};
 const maxFret=Math.max(0,Math.min(MAX_FRET,Math.floor(Number(settings.maxFret??MAX_FRET)||0))),minFret=Math.max(0,Math.min(maxFret,Math.floor(Number(settings.minFret??0)||0))),maxSpan=Math.max(1,Math.min(6,Math.floor(Number(settings.maxSpan??5)||5)));
 const tones=[...new Map(chord.tones.map(t=>[t.pc,{...t}])).values()];
 if(chord.bass!==null&&!tones.some(t=>t.pc===chord.bass))tones.push({pc:chord.bass,note:chord.bassName,degree:0,interval:null,label:'baixo adicional'});
 const highestDegree=Math.max(...tones.map(t=>t.degree)),allowOmissions=settings.allowOmissions!==false,allowRootless=settings.allowRootless!==false;
 const protectedClasses=new Set(chord.tones.filter(t=>t.degree===2||t.degree===4).map(t=>t.pc));
 const optional=tones.filter(t=>!protectedClasses.has(t.pc)&&t.pc!==chord.bass&&((t.degree===1&&allowRootless)||(t.degree===5&&t.interval===7)||(t.degree===11&&t.interval===17&&highestDegree>11)||(t.degree===9&&t.interval===14&&highestDegree>9)));
 const variants=[];const dropCount=Math.max(0,tones.length-4);
 if(!dropCount){variants.push({tones,omitted:[]});if(allowOmissions&&allowRootless&&chord.tones.some(t=>t.degree===7)){const root=tones.find(t=>t.degree===1&&t.pc!==chord.bass);if(root&&tones.length>2)variants.push({tones:tones.filter(t=>t!==root),omitted:[root]});}}
 else if(allowOmissions){function subsets(start,chosen){if(chosen.length===dropCount){const omitted=[...chosen],drop=new Set(omitted.map(t=>t.pc));variants.push({tones:tones.filter(t=>!drop.has(t.pc)),omitted});return;}for(let i=start;i<optional.length;i++)subsets(i+1,[...chosen,optional[i]]);}subsets(0,[]);}
 if(!variants.length)return {chord,results:[],omitted:[],omissionVariants:[],reason:`São ${tones.length} alturas para quatro cordas. Confira as opções de omissão; a terça, a sétima, as alterações e a extensão que define o acorde não são removidas automaticamente.`,settings:{maxFret,minFret,maxSpan}};
 const results=[],seen=new Set();
 for(const variant of variants){const omitted=variant.omitted,active=variant.tones;const rootless=omitted.some(t=>t.degree===1);if(settings.rootPresence==='present'&&rootless||settings.rootPresence==='absent'&&!rootless)continue;
 const target=new Set(active.map(t=>t.pc));const options=tuning.map(n=>{const frets=[];if(settings.allowMuted!==false)frets.push(-1);for(let f=minFret;f<=maxFret;f++)if(target.has((n+f)%12))frets.push(f);return frets;});
 for(const a of options[0])for(const b of options[1])for(const c of options[2])for(const d of options[3]){const frets=[a,b,c,d],pressed=frets.filter(f=>f>0);if(pressed.length&&Math.max(...pressed)-Math.min(...pressed)>=maxSpan)continue;const midi=frets.flatMap((f,i)=>f<0?[]:[f+tuning[i]]);if(midi.length<2)continue;const played=new Set(midi.map(n=>n%12));if(played.size!==target.size)continue;const bass=Math.min(...midi)%12;if(chord.bass!==null&&bass!==chord.bass)continue;if(seen.has(frets.join()))continue;seen.add(frets.join());const fingering=planFingering(frets);const metrics=shapeMetrics(frets,tuning,fingering);const spelling=new Map(chord.tones.map(t=>[t.pc,t.note]));results.push({frets,bass,fingering,metrics,notes:frets.map((f,i)=>f<0?null:spelling.get((f+tuning[i])%12)||names[(f+tuning[i])%12]),omitted,score:metrics.difficulty});}
 }
 sortShapes(results,settings.order||'simple');
 const omitted=results.length?tones.filter(t=>results.every(v=>v.omitted.some(o=>o.pc===t.pc))):[];
 return {chord,results,omitted,omissionVariants:variants.map(v=>v.omitted),reason:results.length?'':`Nenhuma posição encontrada entre as casas ${minFret} e ${maxFret}, com extensão de ${maxSpan} casas. Amplie os filtros; isso não prova que o acorde seja impossível.`,settings:{maxFret,minFret,maxSpan}};
}

## Cifra com desenhos e arranjo por música · 0.13.0

No Repertório, clicar em um acorde abre um desenho junto ao trecho, inclusive em celular, com botão para escolher outra posição. Uma faixa de desenhos mostra as posições usadas na música. **Usar esta forma na música** salva a escolha somente no arranjo daquela cifra, sem alterar a principal global nem outras músicas. Grafias equivalentes compartilham a mesma escolha; transposições têm escolhas separadas. Cifras antigas sem metadados continuam sendo carregadas.

**Preparar formas para cavaquinho** gera uma prévia de posições no instrumento de 12 trastes, com prioridade para registro mais grave/trocas próximas ou menor exigência de dedos. A seleção considera as primeiras aparições dos acordes e um conjunto de candidatos; é uma sugestão determinística, não interpretação de áudio nem garantia de arranjo musical ótimo. Prefere fundamental presente quando disponível. As omissões e os símbolos sem posição reconhecida ficam explícitos. Só após **Aplicar estas formas** as escolhas são registradas, com opção de desfazer. Símbolos harmônicos e texto original são preservados; mudar o tom pela transposição continua sendo uma decisão separada, especialmente para canto.

A grafia brasileira `G#7(4/9)` é interpretada como G#9sus4, com quarta e nona e sem terça. Zero semitons mantém a grafia original, sem trocar G# por Ab na leitura.

As posições e a transposição ficam em `arrangement` em cada registro do armazenamento `mestre-songbook-v1`. Escolhas de formas em músicas já salvas são persistidas automaticamente; alterações no texto precisam de **Salvar cifra**. Para material ainda não salvo, as escolhas são mantidas na tela até salvar. **Baixar cifra com arranjo** gera backup JSON importável, contendo título, texto e escolhas; a exportação TXT mantém o texto original. O PDF continua sendo apenas referência visual.

Validação de navegador: `python test/song_arrangement_browser.py` (Playwright Python e Chromium; servidor ativo; `APP_URL` opcional). Abrange cifra antiga, clique/desenho, escolha por música, recarga, prévia/aplicar/desfazer, preservação do texto, transposição independente, backup e importação, além de dimensões móveis. A cifra pessoal completa não está disponível no servidor: o botão analisa o texto salvo no navegador do aluno.
